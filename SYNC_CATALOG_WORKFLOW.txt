name: Sync 4tochki catalog
on:
  workflow_dispatch:
    inputs:
      dry_run:
        description: 'Check the import without changing the catalog'
        type: boolean
        default: false
      allow_large_drop:
        description: 'Accept a verified decrease of more than 50 percent'
        type: boolean
        default: false
  schedule:
    - cron: '23 2,14 * * *'
permissions:
  contents: write
concurrency:
  group: supplier-catalog-sync
  cancel-in-progress: false
jobs:
  sync:
    if: >-
      github.ref == format('refs/heads/{0}', github.event.repository.default_branch) &&
      (github.event_name == 'workflow_dispatch' || vars.CATALOG_AUTO_SYNC == 'true')
    runs-on: ubuntu-24.04
    timeout-minutes: 50
    steps:
      - uses: actions/checkout@v6
        with:
          ref: ${{ github.event.repository.default_branch }}
      - uses: actions/setup-python@v6
        with:
          python-version: '3.12'
          cache: pip
          cache-dependency-path: requirements-sync.txt
      - uses: actions/setup-node@v6
        with:
          node-version: '22'
          cache: npm
      - name: Install and test the importer
        run: |
          python -m pip install -r requirements-sync.txt
          python -m unittest discover -s tests -p test_catalog_sync.py -v
          python -m unittest discover -s tests -p test_sync_currency.py -v
          python -m unittest discover -s tests -p test_sync_pagination.py -v
          npm ci
          npm test
      - name: Download and validate the catalog
        env:
          FOURTOCHKI_LOGIN: ${{ secrets.FOURTOCHKI_LOGIN }}
          FOURTOCHKI_PASSWORD: ${{ secrets.FOURTOCHKI_PASSWORD }}
          SYNC_DRY_RUN: ${{ inputs.dry_run }}
          SYNC_ALLOW_DROP: ${{ inputs.allow_large_drop }}
        run: |
          git log -1 --format='Source commit: %H'
          args=()
          if [ "$SYNC_DRY_RUN" = "true" ]; then args+=(--dry-run); fi
          if [ "$SYNC_ALLOW_DROP" = "true" ]; then args+=(--allow-large-drop); fi
          python -m backend.catalog_sync "${args[@]}"
      - name: Build the public site
        run: node BUILD_RENDER.mjs
      - name: Commit the verified public snapshot
        if: inputs.dry_run != true
        id: publish
        env:
          CATALOG_BRANCH: ${{ github.event.repository.default_branch }}
        run: |
          git config user.name 'github-actions[bot]'
          git config user.email '41898282+github-actions[bot]@users.noreply.github.com'
          git add -- data/supplier-snapshot.json frontend/assets/products
          if git diff --cached --quiet; then
            echo 'changed=false' >> "$GITHUB_OUTPUT"
          else
            git commit -m 'Update verified supplier catalog'
            git push origin "HEAD:$CATALOG_BRANCH"
            echo 'changed=true' >> "$GITHUB_OUTPUT"
          fi
      - name: Write the run summary
        if: always()
        run: python scripts/sync-summary.py
